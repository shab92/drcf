cmake_minimum_required(VERSION 3.16)
project(drcf_distaead CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(emp-sh2pc REQUIRED)
include_directories(include test)

add_executable(test_ref test/test_ref.cpp)
add_executable(test_distaead test/test_distaead.cpp)
target_link_libraries(test_distaead PRIVATE emp-sh2pc::emp-sh2pc)
add_executable(bench src/bench.cpp)
target_link_libraries(bench PRIVATE emp-sh2pc::emp-sh2pc)
find_package(OpenSSL REQUIRED)
add_executable(test_tsig test/test_tsig.cpp)

target_link_libraries(test_compute PRIVATE emp-sh2pc::emp-sh2pc SEAL::seal)

target_link_libraries(drcf_demo PRIVATE emp-sh2pc::emp-sh2pc SEAL::seal OpenSSL::Crypto)
